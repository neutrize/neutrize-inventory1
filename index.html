GOOGLE APPS SCRIPT — スプレッドシート自動保存
入庫データ・棚卸しデータをGoogleスプレッドシートに自動保存します。
下のGASコードをコピーして設定してください。
GASコード
const SHEET_ID = 'YOUR_SPREADSHEET_ID';
function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.openById(SHEET_ID);
    if (data.type === 'delivery') {
      const sheet = ss.getSheetByName('入庫ログ') || ss.insertSheet('入庫ログ');
      if (sheet.getLastRow() === 0) sheet.appendRow(['日付','仕入先','商品名','カテゴリ','数量','単価','小計']);
      data.items.forEach(item => sheet.appendRow([data.date,data.supplier,item.name,item.category,item.qty,item.price,item.qty*item.price]));
    }
    if (data.type === 'stocktake') {
      const sheet = ss.getSheetByName('棚卸しログ') || ss.insertSheet('棚卸しログ');
      if (sheet.getLastRow() === 0) sheet.appendRow(['日時','商品名','カテゴリ','在庫数','適正量']);
      data.items.forEach(item => sheet.appendRow([new Date(),item.name,item.cat,item.stock,item.target]));
    }
    return ContentService.createTextOutput(JSON.stringify({status:'ok'})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status:'error',message:err.message})).setMimeType(ContentService.MimeType.JSON);
  }
}
function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({status:'ok'})).setMimeType(ContentService.MimeType.JSON);
}
